local function Dragify(frame, parent)
    parent = parent or frame
    local dragging = false
    local dragStartPos, frameStartPos

    frame.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStartPos = i.Position
            frameStartPos = parent.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local delta = i.Position - dragStartPos
            parent.Position = UDim2.new(frameStartPos.X.Scale, frameStartPos.X.Offset + delta.X, frameStartPos.Y.Scale, frameStartPos.Y.Offset + delta.Y)
        end
    end)

    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

local function buttoneffect(frame, entered)
    pcall(function()
        entered.MouseEnter:Connect(function()
            if frame.TextColor3 ~= Color3.fromRGB(234, 239, 246) then
                TweenService:Create(frame, TweenInfo.new(0.06, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {TextColor3 = Color3.fromRGB(234, 239, 245)}):Play()
            end
        end)
        entered.MouseLeave:Connect(function()
            if frame.TextColor3 ~= Color3.fromRGB(157, 171, 182) and frame.TextColor3 ~= Color3.fromRGB(234, 239, 246) then
                TweenService:Create(frame, TweenInfo.new(0.06, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {TextColor3 = Color3.fromRGB(157, 171, 182)}):Play()
            end
        end)
    end)
end

local function clickEffect(button, amount)
    button.MouseButton1Click:Connect(function()
        local new = button.TextSize - tonumber(amount)
        local revert = new + tonumber(amount)
        TweenService:Create(button, TweenInfo.new(0.15, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {TextSize = new}):Play()
        task.wait(0.1)
        TweenService:Create(button, TweenInfo.new(0.1, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {TextSize = revert}):Play()
    end)
end

            return sections
        end
        return tabs
    end

    local WindowObj = {}
    WindowObj.Tabs = tabsections
    WindowObj.Body = Body
    WindowObj.ScreenGui = SG
    return WindowObj
end

local function getRoot()
    local char = LocalPlayer.Character
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart")
end

local function teleportTo(pos)
    local hrp = getRoot()
    if not hrp then return false end
    pcall(function()
        hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
    end)
    return true
end

local function getDumpsterPos()
    if not Refs.Dumpsters then return nil end
    local hrp = getRoot()
    if not hrp then return nil end
    local origin = hrp.Position
    local nearest, minDist = nil, math.huge
    for _, obj in ipairs(Refs.Dumpsters:GetChildren()) do
        if obj:IsA("Model") then
            local pos
            local leaves = obj:FindFirstChild("Leaves")
            if leaves and leaves:IsA("BasePart") then
                pos = leaves.Position
            else
                local ok, bb = pcall(function() return obj:GetBoundingBox() end)
                if ok then pos = bb.Position end
            end
            if pos then
                local d = (pos - origin).Magnitude
                if d < minDist then
                    minDist = d
                    nearest = pos
                end
            end
        end
    end
    return nearest
end

local function getNearestLeafPos()
    if not Refs.LeafFolder or not Refs.LeafSim then return nil end
    local hrp = getRoot()
    if not hrp then return nil end
    local origin = hrp.Position
    local nearest, minDist = nil, math.huge
    local canUseLeaf = Refs.LeafSim.canUseLeaf
    for _, leaf in ipairs(Refs.LeafFolder:GetChildren()) do
        if leaf:IsA("BasePart") then
            local ok, usable = pcall(canUseLeaf, leaf)
            if ok and usable then
                local d = (leaf.Position - origin).Magnitude
                if d < minDist then
                    minDist = d
                    nearest = leaf.Position
                end
            end
        end
    end
    return nearest
end

local function GetCurrentInterval()
    if not Refs.AccelEnabled then
        return Refs.CollectInterval
    end
    local cycle = Refs.AccelDuration + Refs.AccelRestTime
    if cycle <= 0 then return Refs.CollectInterval end
    local pos = (tick() - Refs.AccelCycleStart) % cycle
    if pos < Refs.AccelDuration then
        return Refs.AccelIntervalSpeed
    end
    return Refs.CollectInterval
end

local function RefreshStatus()
    local lbl = Refs.StatusLabel
    if not lbl then return end
    if not Refs.LeafSim then
        lbl.Text = "状态：模块加载失败，请重新执行"
        lbl.TextColor3 = Color3.fromRGB(255, 100, 100)
        return
    end
    local leaves = LocalPlayer:GetAttribute("Leaves") or 0
    local capacity = LocalPlayer:GetAttribute("LeafCapacity") or 25
    if Refs.UserIntent then
        if Refs.PausedByBag then
            if Refs.AutoSell then
                lbl.Text = "状态：自动出售中..."
                lbl.TextColor3 = Color3.fromRGB(255, 200, 100)
            else
                lbl.Text = "状态：背包已满，请出售"
                lbl.TextColor3 = Color3.fromRGB(255, 100, 100)
            end
        else
            local accelTag = ""
            if Refs.AccelEnabled then
                local cycle = Refs.AccelDuration + Refs.AccelRestTime
                if cycle > 0 then
                    local pos = (tick() - Refs.AccelCycleStart) % cycle
                    if pos < Refs.AccelDuration then
                        accelTag = " [加速中]"
                    end
                end
            end
            lbl.Text = "状态：自动收集中 (" .. leaves .. "/" .. capacity .. ")" .. accelTag
            lbl.TextColor3 = Color3.fromRGB(100, 255, 100)
        end
    else
        lbl.Text = "状态：准备就绪 (" .. leaves .. "/" .. capacity .. ")"
        lbl.TextColor3 = Color3.fromRGB(200, 200, 200)
    end
end

local function OnLeavesChanged()
    local leaves = LocalPlayer:GetAttribute("Leaves") or 0
    local capacity = LocalPlayer:GetAttribute("LeafCapacity") or 25
    if leaves >= capacity then
        if Refs.UserIntent and not Refs.PausedByBag then
            Refs.PausedByBag = true
            if not Refs.AutoSell then
                Notify("自动捡树叶", "背包已满，自动暂停。请出售。")
            end
        end
    else
        if Refs.PausedByBag then
            Refs.PausedByBag = false
        end
    end
    RefreshStatus()
end

local sharedOverlapParams = OverlapParams.new()
sharedOverlapParams.FilterType = Enum.RaycastFilterType.Include
local overlapFilterSet = false

task.spawn(function()
    while true do
        task.wait(GetCurrentInterval())

        if not Refs.UserIntent then continue end
        if not Refs.LeafSim then continue end

        if Refs.PausedByBag and Refs.AutoSell and not Refs.IsTeleporting then
            Refs.IsTeleporting = true
            local dumpPos = getDumpsterPos()
            local hrp = getRoot()
            if dumpPos and hrp then
                local savedCFrame = hrp.CFrame
                teleportTo(dumpPos)
                task.wait(Refs.SellDelay)
                pcall(function() Refs.EmptyBackpack:FireServer() end)
                task.wait(Refs.SellDelay)
                pcall(function()
                    local hrp2 = getRoot()
                    if hrp2 then hrp2.CFrame = savedCFrame end
                end)
            end
            task.wait(Refs.TpCooldown)
            Refs.IsTeleporting = false
            continue
        end

        if Refs.PausedByBag then continue end

        local hrp = getRoot()
        if not hrp then continue end

        if not Refs.IsTeleporting then
            Refs.OriginCFrame = hrp.CFrame
        end

        if not overlapFilterSet and Refs.LeafFolder then
            sharedOverlapParams.FilterDescendantsInstances = { Refs.LeafFolder }
            overlapFilterSet = true
        end

        local current = LocalPlayer:GetAttribute("Leaves") or 0
        local capacity = LocalPlayer:GetAttribute("LeafCapacity") or 25
        local space = capacity - current
        if space <= 0 then continue end

        local parts = workspace:GetPartBoundsInRadius(hrp.Position, Refs.ScanRadius, sharedOverlapParams)
        local batch = {}
        local maxCount = math.min(space, Refs.MaxBatch)
        local canUseLeaf = Refs.LeafSim.canUseLeaf

        for _, part in ipairs(parts) do
            if #batch >= maxCount then break end
            if part:IsA("BasePart") then
                local ok, usable = pcall(canUseLeaf, part)
                if ok and usable then
                    table.insert(batch, part)
                end
            end
        end

        if #batch > 0 then
            pcall(function() Refs.CollectMany(batch) end)
        elseif Refs.AutoFarm then
            local leafPos = getNearestLeafPos()
            if leafPos then
                Refs.IsTeleporting = true
                teleportTo(leafPos)
                task.wait(Refs.TpCooldown)
                Refs.IsTeleporting = false
            end
        end
    end
end)
