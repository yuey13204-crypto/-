local function gradient(parent, c1, c2, rot)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(c1, c2)
    g.Rotation = rot or 90
    g.Parent = parent
    return g
end

local function gradient3(parent, c1, c2, c3, rot)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new{
        ColorSequenceKeypoint.new(0, c1),
        ColorSequenceKeypoint.new(0.5, c2),
        ColorSequenceKeypoint.new(1, c3),
    }
    g.Rotation = rot or 90
    g.Parent = parent
    return g
end

local function stroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.Stroke
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = parent
    return s
end

local function corner(parent, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = (type(r) == "number") and UDim.new(0, r) or r
    c.Parent = parent
    return c
end

local function makeText(parent, props)
    local t = Instance.new("TextLabel")
    t.BackgroundTransparency = 1
    t.BorderSizePixel = 0
    t.Font = props.font or Enum.Font.Gotham
    t.Text = props.text or ""
    t.TextColor3 = props.color or Theme.Text
    t.TextSize = props.size or 13
    t.TextXAlignment = props.xAlign or Enum.TextXAlignment.Left
    t.TextYAlignment = Enum.TextYAlignment.Center
    t.AnchorPoint = props.anchor or Vector2.new(0, 0.5)
    t.Position = props.position or UDim2.new(0, 0, 0.5, 0)
    t.Size = props.sizeUD or UDim2.new(0, 100, 0, 20)
    t.TextTruncate = props.truncate or Enum.TextTruncate.None
    t.Parent = parent
    return t
end

local function Dragify(handle, target)
    target = target or handle
    local dragging, startPos, startParent = false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            startPos = i.Position
            startParent = target.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local delta = i.Position - startPos
            target.Position = UDim2.new(
                startParent.X.Scale, startParent.X.Offset + delta.X,
                startParent.Y.Scale, startParent.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

    local function FindPlayerVehicle()
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return nil end
        local seat = hum.SeatPart
        if not seat then return nil end

    local function GetPlayerSeat()
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass("Humanoid")
        return hum and hum.SeatPart
    end

    local function ScanVehicle()
        local v = FindPlayerVehicle()
        if v == STATE.CurrentVehicle then return end
        STATE.CurrentVehicle = v
        STATE.FuelProps = {}
        if not v then return end

        for _, obj in ipairs(v:GetDescendants()) do
            if obj:IsA("NumberValue") or obj:IsA("IntValue") then
                local n = string.lower(obj.Name)
                if string.find(n, "fuel") or string.find(n, "gas") then
                    table.insert(STATE.FuelProps, {Type = "Value", Obj = obj})
                end
            elseif obj:IsA("Configuration") then
                for _, sub in ipairs(obj:GetChildren()) do
                    if sub:IsA("NumberValue") or sub:IsA("IntValue") then
                        local n = string.lower(sub.Name)
                        if string.find(n, "fuel") or string.find(n, "gas") then
                            table.insert(STATE.FuelProps, {Type = "Value", Obj = sub})
                        end
                    end
                end
            end
        end

        pcall(function()
            for name, val in pairs(v:GetAttributes()) do
                local ln = string.lower(name)
                if (string.find(ln, "fuel") or string.find(ln, "gas")) and type(val) == "number" then
                    table.insert(STATE.FuelProps, {Type = "Attr", Obj = v, Key = name})
                end
            end
        end)

        local seat = GetPlayerSeat()
        if seat then
            pcall(function()
                for name, val in pairs(seat:GetAttributes()) do
                    local ln = string.lower(name)
                    if (string.find(ln, "fuel") or string.find(ln, "gas")) and type(val) == "number" then
                        table.insert(STATE.FuelProps, {Type = "Attr", Obj = seat, Key = name})
                    end
                end
            end)
        end
    end

    local function ScanFuelInGC()
        pcall(function()
            for _, obj in pairs(getgc(true)) do
                if type(obj) == "table" then
                    for k, v in pairs(obj) do
                        if type(k) == "string" then
                            local lk = string.lower(k)
                            if (string.find(lk, "fuel") or string.find(lk, "gas"))
                                and type(v) == "number"
                                and v >= 0 and v < 999999 then
                                pcall(function() obj[k] = CONFIG.FuelValue end)
                            end
                        end
                    end
                end
            end
        end)
    end

    local function ApplyFuel()
        local v = STATE.CurrentVehicle
        if not v then return end
        local val = CONFIG.FuelValue
        for _, entry in ipairs(STATE.FuelProps) do
            if entry.Type == "Value" then
                local obj = entry.Obj
                if obj and obj.Parent and obj.Value ~= val then
                    pcall(function() obj.Value = val end)
                end
            elseif entry.Type == "Attr" then
                pcall(function()
                    if entry.Obj:GetAttribute(entry.Key) ~= val then
                        entry.Obj:SetAttribute(entry.Key, val)
                    end
                end)
            end
        end
        pcall(function()
            for name, val2 in pairs(v:GetAttributes()) do
                local ln = string.lower(name)
                if (string.find(ln, "fuel") or string.find(ln, "gas")) and type(val2) == "number" then
                    v:SetAttribute(name, val)
                end
            end
        end)
    end

        local ignoreDead = CONFIG.IgnoreDead
        local pool = STATE._pool
        local poolSize = 0
        local npcCount = #NPC_ARRAY

        for i = 1, npcCount do
            local npc = NPC_ARRAY[i]
            local entry = NPC_ENTRIES[npc]
            if entry then

                            local dx = mx - rp.X
                            local dy = my - rp.Y
                            local dz = mz - rp.Z
                            local dSq = dx * dx + dy * dy + dz * dz
                            if dSq <= rangeSq then
                                poolSize = poolSize + 1
                                local slot = pool[poolSize]
                                if not slot then slot = {}; pool[poolSize] = slot end
                                slot.npc = npc
                                slot.dSq = dSq
                            end
                        end
                    end
                end
            end
        end

        if poolSize == 0 then STATE.CurrentTarget = nil; return STATE._result, 0 end

        local take = math.min(CONFIG.MaxTargets, poolSize)
        local result = STATE._result
        local mode = CONFIG.SortMode

        if mode == "Random" then
            for i = 1, take do
                local j = math.random(i, poolSize)
                pool[i], pool[j] = pool[j], pool[i]
                result[i] = pool[i].npc
            end
        else
            local isNear = (mode == "Near")
            for i = 1, take do
                local bestIdx = i
                local bestVal = pool[i].dSq
                for j = i + 1, poolSize do
                    local val = pool[j].dSq
                    if (isNear and val < bestVal) or (not isNear and val > bestVal) then
                        bestIdx = j
                        bestVal = val
                    end
                end
                if bestIdx ~= i then pool[i], pool[bestIdx] = pool[bestIdx], pool[i] end
                result[i] = pool[i].npc
            end
        end

        result[take + 1] = nil
        STATE.CurrentTarget = result[1]
        return result, take
    end

    local function GetSafeTarget()
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end
        local myPos = hrp.Position
        local mx, my, mz = myPos.X, myPos.Y, myPos.Z
        local range = CONFIG.SafeRange
        local rangeSq = range * range
        local ignoreBroken = CONFIG.SafeIgnoreBroken
        local safes = GetSafeList(false)
        local best, bestDSq = nil, rangeSq
        for i = 1, #safes do
            local safe = safes[i]
            local entry = SAFE_ENTRIES[safe]
            if entry then
                local root = entry.Root
                if root.Parent then
                    if not ignoreBroken or not IsSafeBroken(safe) then
                        local rp = root.Position
                        local dx = mx - rp.X
                        local dy = my - rp.Y
                        local dz = mz - rp.Z
                        local dSq = dx * dx + dy * dy + dz * dz
                        if dSq < bestDSq then bestDSq = dSq; best = safe end
                    end
                end
            end
        end
        STATE.CurrentSafe = best
        return best
    end

    local function ComputeTPCFrame(hrp, targetRoot)
        local targetPos = targetRoot.Position + Vector3.new(0, CONFIG.TP_HeightOffset, 0)
        local fromPos = hrp.Position
        local dir = fromPos - targetPos
        local mag = dir.Magnitude
        if mag < 0.1 then dir = Vector3.new(0, 0, 1) else dir = dir / mag end
        local tpPos = targetPos + dir * CONFIG.TP_Distance
        return CFrame.lookAt(tpPos, Vector3.new(targetPos.X, tpPos.Y, targetPos.Z))
    end

    local function CheckTeleportPath(fromPos, toPos, char)
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {char}
        params.IgnoreWater = true
        local dir = toPos - fromPos
        local result = Workspace:Raycast(fromPos, dir, params)
        if not result then return true end
        if result.Distance >= (dir.Magnitude - 3) then return true end
        return false
    end

    local function TPAttackWith(npcRoot, fireFn)
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return false end
        if not npcRoot or not npcRoot.Parent then return false end
        local origCF = hrp.CFrame
        local origVel = hrp.AssemblyLinearVelocity
        local origPos = origCF.Position
        local tpCF = ComputeTPCFrame(hrp, npcRoot)

        if CONFIG.TP_BackupDirs then
            if not CheckTeleportPath(origPos, tpCF.Position, char) then
                local altDirs = {
                    Vector3.new(1, 0, 0), Vector3.new(-1, 0, 0),
                    Vector3.new(0, 0, 1), Vector3.new(0, 0, -1),
                    Vector3.new(0, 1, 0),
                }
                local basePos = npcRoot.Position + Vector3.new(0, CONFIG.TP_HeightOffset, 0)
                for _, d in ipairs(altDirs) do
                    local altPos = basePos + d * CONFIG.TP_Distance
                    local altCF = CFrame.lookAt(altPos, basePos)
                    if CheckTeleportPath(origPos, altPos, char) then tpCF = altCF; break end
                end
            end
        end

        if CONFIG.TP_VisualFix then
            EnsureVisualFix()
            STATE._visualCF = origCF
        end

        hrp.CFrame = tpCF
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
        RunService.Heartbeat:Wait()

        local ok = fireFn()

        if CONFIG.TP_HoldFrames > 1 then
            for _ = 2, CONFIG.TP_HoldFrames do RunService.Heartbeat:Wait() end
        end

        hrp.CFrame = origCF
        hrp.AssemblyLinearVelocity = origVel
        STATE._visualCF = nil
        return ok
    end

    local function AttackNPC(npc)
        local entry = NPC_ENTRIES[npc]
        if not entry or not entry.Hum.Parent then return false end
        if CONFIG.IgnoreDead and entry.Hum.Health <= 0 then return false end
        if CONFIG.RangeMode == "TP" then
            return TPAttackWith(entry.Root, function() return FireNPCDamage(entry.Hum) end)
        else
            return FireNPCDamage(entry.Hum)
        end
    end

    local function AttackSafe(safe)
        local entry = SAFE_ENTRIES[safe]
        if not entry or not entry.Root.Parent then return false end
        if CONFIG.SafeIgnoreBroken and IsSafeBroken(safe) then return false end
        if CONFIG.RangeMode == "TP" then
            return TPAttackWith(entry.Root, function() return FireSafeDamage(safe) end)
        else
            return FireSafeDamage(safe)
        end
    end

    local LP = LocalPlayer

    local CONFIG = {
        Enabled = true,
        Mode = "First",
        ThirdPersonMinZoom = 8,
        ThirdPersonMaxZoom = 20,
        ThirdPersonDefault = 14,
        HideCameraPosition = true,
        LockThirdPerson = true,
    }

    local hasHook = type(hookmetamethod) == "function"
        and type(getrawmetatable) == "function"
        and type(newcclosure) == "function"
        and type(setreadonly) == "function"

    local gameMeta = hasHook and getrawmetatable(game) or nil
    local originalIndex = nil
    local hookInstalled = false

    local function installHook()
        if hookInstalled then return end
        if not hasHook or not CONFIG.HideCameraPosition then return end
        if not originalIndex then originalIndex = gameMeta.__index end
        setreadonly(gameMeta, false)
        gameMeta.__index = newcclosure(function(self, key)
            if CONFIG.Mode == "Third" and CONFIG.Enabled and self == Camera and key == "CFrame" then
                local real = originalIndex(Camera, "CFrame")
                local char = LP.Character
                local head = char and char:FindFirstChild("Head")
                if head and real then
                    return CFrame.new(head.Position, head.Position + real.LookVector * 100)
                end
            end
            return originalIndex(self, key)
        end)
        setreadonly(gameMeta, true)
        hookInstalled = true
    end

    local function uninstallHook()
        if not hookInstalled then return end
        if not hasHook then return end
        setreadonly(gameMeta, false)
        gameMeta.__index = originalIndex
        setreadonly(gameMeta, true)
        hookInstalled = false
    end

    local function restoreCameraSettings()
        pcall(function()
            LP.CameraMode = Enum.CameraMode.Classic
            LP.CameraMinZoomDistance = 0.5
            LP.CameraMaxZoomDistance = 400
        end)
    end

    local function applyThirdPerson()
        pcall(function()
            LP.CameraMode = Enum.CameraMode.Classic
            LP.CameraMinZoomDistance = CONFIG.ThirdPersonMinZoom
            LP.CameraMaxZoomDistance = CONFIG.ThirdPersonMaxZoom
        end)
        installHook()
    end

    local function applyFirstPerson()
        uninstallHook()
        restoreCameraSettings()
    end

    local function setMode(mode)
        CONFIG.Mode = mode
        if mode == "First" then
            applyFirstPerson()
        else
            applyThirdPerson()
        end
    end

        if not CONFIG.LockThirdPerson then return end
        if CONFIG.Mode ~= "Third" then return end
        if CONFIG.HideCameraPosition and not hookInstalled then installHook() end
        if LP.CameraMode == Enum.CameraMode.LockFirstPerson then

        end

        end
    end)
